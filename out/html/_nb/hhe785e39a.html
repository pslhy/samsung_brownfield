67// Rule 3 (InlineVariable):
68// int n = e; ... for (i = 0; i < n; i++) ... ~> ... for (i = 0; i < e; i++) ...
69// Condition: the value of e does not change while the variable is used.
70//
71// The place where it breaks in HomeHub: Dispatcher.evictOldestNonCritical rotates the queue once,
72// keeping every event except the oldest non-critical one (the victim):
73// int n = queue.size();
74// for (int i = 0; i < n; i++) { x = queue.poll(); if (no victim yet && !x.critical) victim = x; else queue.offer(x); }
75// Inlining n turns the bound into queue.size(), which shrinks when the victim is taken out.
76module InlineVariableRule {
77 datatype Event = Event(id: int, critical: bool)
78 datatype State = State(queue: seq<Event>, victim: int) // victim: id, or 0 for none
79
80 // one iteration of the loop body (the queue is never empty inside the loop)
81 function Step(s: State): State
82 requires |s.queue| > 0
83 {
84 var x := s.queue[0];
85 if s.victim == 0 && !x.critical then State(s.queue[1..], x.id)
86 else State(s.queue[1..] + [x], s.victim)
87 }
88
89 // with the variable: the bound n is fixed when the loop starts
90 function Fixed(s: State, i: nat, n: nat): State
91 decreases n - i
92 {
93 if i >= n || |s.queue| == 0 then s else Fixed(Step(s), i + 1, n)
94 }
95
96 // inlined: the bound is re-read from the queue every time
97 function Inlined(s: State, i: nat, fuel: nat): State
98 decreases fuel
99 {
100 if fuel == 0 || i >= |s.queue| then s else Inlined(Step(s), i + 1, fuel - 1)
101 }
102
103 predicate AllCritical(q: seq<Event>) { forall k :: 0 <= k < |q| ==> q[k].critical }
104
105 lemma StepKeepsSize(s: State)
106 requires |s.queue| > 0 && AllCritical(s.queue)
107 ensures |Step(s).queue| == |s.queue| && AllCritical(Step(s).queue)
108 {
109 }
110
111 lemma Loops(s: State, i: nat, n: nat, fuel: nat)
112 requires AllCritical(s.queue) && |s.queue| == n && i <= n && fuel >= n - i
113 ensures Fixed(s, i, n) == Inlined(s, i, fuel)
114 decreases n - i
115 {
116 if i < n && |s.queue| > 0 {
117 StepKeepsSize(s);
118 Loops(Step(s), i + 1, n, fuel - 1);
119 }
120 }
121
122 // Condition met: nothing is taken out of the queue inside the loop (here: every event is
123 // critical, so there is no victim), so queue.size() stays n and the two loops are the same.
124 lemma InlineVariable_Same(q: seq<Event>)
125 requires AllCritical(q) // condition: the loop does not change queue.size()
126 ensures Fixed(State(q, 0), 0, |q|) == Inlined(State(q, 0), 0, |q|)
127 {
128 Loops(State(q, 0), 0, |q|, |q|);
129 }
130
131 // Without the condition: three non-critical events 1, 2, 3. With the variable, 1 is taken out and
132 // 2, 3 keep their order. Inlined, the loop stops one round early and 3 jumps ahead of 2.
133 lemma InlineVariable_Counterexample()
134 ensures var q := [Event(1, false), Event(2, false), Event(3, false)];
135 Fixed(State(q, 0), 0, 3).queue == [Event(2, false), Event(3, false)] &&
136 Inlined(State(q, 0), 0, 3).queue == [Event(3, false), Event(2, false)]
137 {
138 }
139}
