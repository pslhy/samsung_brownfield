20// ---------------------------------------------------------------------------------------------본문 규칙 ②. 반복의 몸체는 "(지금 원소, 지금 목록) → 몸체가 끝난 뒤의 목록" 함수 body로 나타냄. 두 반복이 어떤 원소를 어떤 순서로 방문하는지 비교
21// Rule 2 (ForEach):
22// for (int i = 0; i < list.size(); i++) { body(list.get(i)); } ~> for (T x : list) { body(x); }
23// Condition: the body does not change the list.
24//
25// The body is modeled as a function from (element, current list) to the list after the body.
26// We compare the elements the two loops visit, in order.
27module ForEachRule {본문
28 // index loop: re-reads list.size() and list.get(i) on the CURRENT list after each body함수 바꾸기 전: 인덱스 반복. 매번 지금 목록에서 i번째를 읽고, 몸체가 바꾼 목록으로 다음 반복을 함. 방문한 원소 목록을 돌려줌. fuel은 반복이 끝난다는 근거
29 function IndexVisits(list: seq<int>, body: (int, seq<int>) -> seq<int>, i: nat, fuel: nat): seq<int>
30 decreases fuel
31 {
32 if fuel == 0 || i >= |list| then []
33 else [list[i]] + IndexVisits(body(list[i], list), body, i + 1, fuel - 1)
34 }
35 
36 // for-each: visits the elements of the list as it was when the loop started함수 바꾼 뒤: for-each. 시작할 때의 목록을 처음부터 끝까지 방문
37 function EachVisits(list: seq<int>): seq<int> { list }
38 
39 lemma Visits(list: seq<int>, body: (int, seq<int>) -> seq<int>, i: nat, fuel: nat)증명 도우미 증명 도우미: i번째부터 방문한 것이 목록의 i번째 이후와 같다는 것을 하나씩 늘려 가며 증명
40 requires forall x, l :: body(x, l) == l // condition
41 requires i <= |list| && fuel >= |list| - i
42 ensures IndexVisits(list, body, i, fuel) == list[i..]
43 decreases |list| - i
44 {
45 if i < |list| {
46 Visits(list, body, i + 1, fuel - 1);
47 }
48 }

50 lemma ForEach_Same(list: seq<int>, body: (int, seq<int>) -> seq<int>)본문 같음 보조정리
51 requires forall x, l :: body(x, l) == l // condition: body does not change the list사전조건 조건: 몸체가 목록을 바꾸지 않음. 어떤 원소, 어떤 목록을 받아도 목록을 그대로 돌려줌
52 ensures IndexVisits(list, body, 0, |list|) == EachVisits(list)사후조건 증명할 것: 두 반복이 방문하는 원소가 같음
53 {증명 도우미 위 도우미를 불러 증명
54 Visits(list, body, 0, |list|);
55 }

57 // Without the condition: a body that removes the first element. [1, 2, 3]: the index loop본문 반례 보조정리: 조건이 없으면 실제로 다르다는 구체적인 예. 같음 보조정리는 "조건이 있으면 같다"만 말하므로, 이것이 있어야 "조건이 꼭 필요하다"고 말할 수 있음
58 // visits 1 and 3 and skips 2.
59 lemma ForEach_Counterexample()사후조건 몸체가 맨 앞 원소를 지우면, [1, 2, 3]에서 인덱스 반복은 1, 3을 방문하고(2를 건너뜀) for-each는 1, 2, 3을 방문
60 ensures var removeFirst := (x: int, l: seq<int>) => if |l| > 0 then l[1..] else l;
61 IndexVisits([1, 2, 3], removeFirst, 0, 3) == [1, 3] && EachVisits([1, 2, 3]) == [1, 2, 3]
