67// Rule 3 (InlineVariable):
68// int n = e; ... for (i = 0; i < n; i++) ... ~> ... for (i = 0; i < e; i++) ...
69// Condition: the value of e does not change while the variable is used.
70//
71// The place where it breaks in HomeHub: Dispatcher.evictOldestNonCritical rotates the queue once,
72// keeping every event except the oldest non-critical one (the victim):
73// int n = queue.size();
74// for (int i = 0; i < n; i++) { x = queue.poll(); if (no victim yet && !x.critical) victim = x; else queue.offer(x); }
75// Inlining n turns the bound into queue.size(), which shrinks when the victim is taken out.
76module InlineVariableRule {Java 필드 이벤트(번호, 중요 여부)와 반복의 상태(대기열 내용, 밀어낸 이벤트 번호. 없으면 0)
77 datatype Event = Event(id: int, critical: bool)
78 datatype State = State(queue: seq<Event>, victim: int) // victim: id, or 0 for none

80 // one iteration of the loop body (the queue is never empty inside the loop)함수 반복 한 번: 맨 앞을 꺼내, 아직 밀어낸 것이 없고 비중요면 밀어냄. 아니면 뒤에 다시 넣음(Java의 poll/offer)
81 function Step(s: State): State
82 requires |s.queue| > 0
83 {
84 var x := s.queue[0];
85 if s.victim == 0 && !x.critical then State(s.queue[1..], x.id)
86 else State(s.queue[1..] + [x], s.victim)
87 }

98 decreases fuel
99 {
100 if fuel == 0 || i >= |s.queue| then s else Inlined(Step(s), i + 1, fuel - 1)
