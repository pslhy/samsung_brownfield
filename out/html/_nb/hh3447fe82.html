66// ---------------------------------------------------------------------------------------------본문 규칙 ③. 상태는 무엇이든 됨(타입 S). 식 e는 상태의 함수 size, 반복 몸체는 "상태 → 몸체가 끝난 뒤의 상태" 함수 body로 나타냄. 두 반복이 끝난 뒤의 상태를 비교
67// Rule 3 (InlineVariable):
68// int n = e; for (int i = 0; i < n; i++) { body } ~> for (int i = 0; i < e; i++) { body }
69// Condition: the body does not change the value of e.
70//
71// The state can be anything (type S). The expression e is a function `size` of the state, and the
72// loop body a function from state to state. We compare the states the two loops end in.
73module InlineVariableRule {본문
74 // with the variable: n is computed once, before the loop함수 바꾸기 전: n을 반복 전에 한 번 계산해 두고 n번 돎(int n = e;)
75 function Fixed<S>(s: S, body: S -> S, i: nat, n: nat): S
76 decreases n - i
77 {
78 if i >= n then s else Fixed(body(s), body, i + 1, n)
79 }
80 
81 // inlined: e (here size(s)) is computed again before every round, on the current state함수 바꾼 뒤: 매번 지금 상태에서 size를 다시 계산해 비교(i < e). fuel은 반복이 끝난다는 근거
82 function Inlined<S>(s: S, size: S -> nat, body: S -> S, i: nat, fuel: nat): S
83 decreases fuel
84 {
85 if fuel == 0 || i >= size(s) then s else Inlined(body(s), size, body, i + 1, fuel - 1)
86 }
87 
88 lemma Loops<S>(s: S, size: S -> nat, body: S -> S, i: nat, n: nat, fuel: nat)증명 도우미 증명 도우미: 몸체가 size를 바꾸지 않으면 두 반복이 한 단계씩 같게 진행됨을 하나씩 늘려 가며 증명
89 requires forall t :: size(body(t)) == size(t)
90 requires size(s) == n && i <= n && fuel >= n - i
91 ensures Fixed(s, body, i, n) == Inlined(s, size, body, i, fuel)
92 decreases n - i
93 {
94 if i < n {
95 Loops(body(s), size, body, i + 1, n, fuel - 1);
96 }
97 }
98 
99 lemma InlineVariable_Same<S>(s: S, size: S -> nat, body: S -> S)본문 같음 보조정리
100 requires forall t :: size(body(t)) == size(t) // condition: the body does not change e사전조건 조건: 몸체가 식의 값을 바꾸지 않음. 어떤 상태 t에서도 몸체를 한 번 돈 뒤의 size가 그대로
101 ensures Fixed(s, body, 0, size(s)) == Inlined(s, size, body, 0, size(s))사후조건 증명할 것: 처음 size(s)번 도는 반복과, 매번 size와 비교하는 반복이 끝난 뒤의 상태가 같음
102 {증명 도우미 위 도우미(Loops)를 불러 증명
103 Loops(s, size, body, 0, size(s), size(s));
104 }
105 
106 // Without the condition: the state is a list, e is its length, and the body takes out the first본문 반례 보조정리: 조건이 없으면 실제로 다르다는 구체적인 예. 이것이 있어야 "조건이 꼭 필요하다"고 말할 수 있음
107 // element. [1, 2, 3]: with the variable the loop runs 3 times and ends at []; inlined, it stops
108 // after 2 rounds at [3] (the loop ends one round early, as in lab E-2).
109 lemma InlineVariable_Counterexample()
110 ensures var size := (l: seq<int>) => |l|;사후조건 상태는 목록, 식은 목록 길이, 몸체는 맨 앞을 하나 뺌. [1, 2, 3]에서 변수를 쓰면 3번 돌아 [], 인라인하면 [2, 3] → [3]에서 2 < 1이 거짓이라 멈춰 [3]. E-2에서 반복이 한 번 일찍 끝난 것과 같은 일
111 var removeFirst := (l: seq<int>) => if |l| > 0 then l[1..] else l;
112 Fixed([1, 2, 3], removeFirst, 0, 3) == [] &&
113 Inlined([1, 2, 3], size, removeFirst, 0, 3) == [3]
